cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0149)
  cmake_policy(SET CMP0149 OLD)
endif()
project(DesktopPdfIntegration LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(IntegrationContract tests/Contract.cpp)
add_test(NAME integration-portable-contract COMMAND IntegrationContract)
if(NOT WIN32)
  message(STATUS "Portable production-helper checks only; NO Windows API or integration evidence.")
  return()
endif()
if(NOT MSVC OR NOT CMAKE_SYSTEM_VERSION STREQUAL "10.0.26100.0"
   OR NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL "10.0.26100.0"
   OR NOT CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|ARM64)$")
  message(FATAL_ERROR "Native x64/ARM64 MSVC and explicitly selected SDK 26100 required.")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(IntegrationWorkerProcess STATIC WorkerProcess.cpp)
add_library(IntegrationTask SHARED Task.cpp Task.def)
add_executable(IntegrationApp WIN32 ../printer/psa/App.cpp)
add_executable(IntegrationHost WIN32 Host.cpp)
add_executable(IntegrationClient Client.cpp)
add_executable(IntegrationNative tests/Native.cpp)
add_executable(IntegrationProcessTests tests/Process.cpp)
add_executable(IntegrationWorkerFixture tests/WorkerFixture.cpp)
foreach(target IntegrationContract IntegrationWorkerProcess IntegrationTask IntegrationApp IntegrationHost IntegrationClient IntegrationNative IntegrationProcessTests IntegrationWorkerFixture)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /utf-8)
  target_link_libraries(${target} PRIVATE windowsapp ole32)
endforeach()
add_test(NAME integration-native-record-status COMMAND IntegrationNative)
target_link_libraries(IntegrationProcessTests PRIVATE IntegrationWorkerProcess)
foreach(case success crash deadline cancel precancel malformed duplicate trailing missing mismatch exit24 exit71 descendant breakaway handles channel killclose publish existing race pubcancel journal validation)
  add_test(NAME integration-production-${case}
    COMMAND IntegrationProcessTests ${case} $<TARGET_FILE:IntegrationWorkerFixture>)
  set_tests_properties(integration-production-${case} PROPERTIES TIMEOUT 60)
endforeach()
if(CMAKE_GENERATOR_PLATFORM STREQUAL "ARM64")
  set(PDF_PACKAGE_ARCH arm64)
else()
  set(PDF_PACKAGE_ARCH x64)
endif()
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/package/AppxManifest.xml" PDF_MANIFEST)
string(REPLACE "ProcessorArchitecture=\"x64\"" "ProcessorArchitecture=\"${PDF_PACKAGE_ARCH}\""
  PDF_MANIFEST "${PDF_MANIFEST}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/AppxManifest.xml" "${PDF_MANIFEST}")
install(TARGETS IntegrationApp IntegrationTask IntegrationHost IntegrationClient RUNTIME DESTINATION . LIBRARY DESTINATION .)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/AppxManifest.xml" DESTINATION .)
install(FILES ../printer/package/DiagnosticPdc.xml DESTINATION Config RENAME IntegrationPdc.xml)
install(FILES package/Strings/en-US/Resources.resw DESTINATION Strings/en-US)
target_link_libraries(IntegrationWorkerProcess PUBLIC windowsapp ole32 shell32 advapi32)
target_link_libraries(IntegrationHost PRIVATE IntegrationWorkerProcess)
target_link_libraries(IntegrationClient PRIVATE IntegrationWorkerProcess)
foreach(target IntegrationTask IntegrationApp)
  target_compile_definitions(${target} PRIVATE PDF_APP WINAPI_FAMILY=WINAPI_FAMILY_APP)
  target_link_options(${target} PRIVATE /APPCONTAINER)
endforeach()
