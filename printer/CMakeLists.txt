cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0149)
  cmake_policy(SET CMP0149 OLD)
endif()
project(PsaFeasibility LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Build with MSVC and Windows SDK 10.0.26100.0 on Windows.")
endif()
if(NOT CMAKE_SYSTEM_VERSION STREQUAL "10.0.26100.0")
  message(FATAL_ERROR "Select Windows SDK 10.0.26100.0 explicitly.")
endif()
if(NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL "10.0.26100.0")
  message(FATAL_ERROR "The selected Visual Studio instance must use SDK 10.0.26100.0.")
endif()
if(NOT CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|ARM64)$")
  message(FATAL_ERROR "Select the x64 or ARM64 target explicitly (-A).")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

add_executable(PrintProbe probe/PrintProbe.cpp)
target_compile_definitions(PrintProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(PrintProbe PRIVATE /W4 /WX /EHsc)
target_link_libraries(PrintProbe PRIVATE gdi32 winspool advapi32 ole32)

add_library(PsaTask SHARED psa/Task.cpp psa/Task.def)
add_executable(PsaDiagnostic WIN32 psa/App.cpp)
foreach(target PsaTask PsaDiagnostic)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN
    NOMINMAX WINAPI_FAMILY=WINAPI_FAMILY_APP)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive-)
  target_link_options(${target} PRIVATE /APPCONTAINER)
  target_link_libraries(${target} PRIVATE windowsapp)
endforeach()

enable_testing()
add_test(NAME probe-arguments COMMAND PrintProbe --self-test)
add_executable(ActivationProbe tests/ActivationProbe.cpp)
target_compile_definitions(ActivationProbe PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(ActivationProbe PRIVATE /W4 /WX /EHsc)
target_link_libraries(ActivationProbe PRIVATE windowsapp)
add_dependencies(ActivationProbe PsaTask)
add_test(NAME task-factory COMMAND ActivationProbe $<TARGET_FILE:PsaTask>)
