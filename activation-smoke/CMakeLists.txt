cmake_minimum_required(VERSION 3.24)
if(POLICY CMP0149)
  cmake_policy(SET CMP0149 OLD)
endif()
project(PsaActivationSmoke LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SYSTEM_VERSION STREQUAL "10.0.26100.0"
   OR NOT CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION STREQUAL "10.0.26100.0")
  message(FATAL_ERROR "MSVC and explicitly selected Windows SDK 10.0.26100.0 required.")
endif()
if(NOT CMAKE_GENERATOR_PLATFORM MATCHES "^(x64|ARM64)$")
  message(FATAL_ERROR "Select native x64 or ARM64 explicitly.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(SmokeTask SHARED Task.cpp Task.def)
add_executable(SmokeApp WIN32 ../printer/psa/App.cpp)
add_executable(SmokeHost WIN32 Host.cpp)
add_executable(SmokeObserver Observer.cpp)
add_executable(SmokeTests tests/Native.cpp)
add_executable(SmokeIdentifiers tests/Identifiers.cpp)
foreach(target SmokeTask SmokeApp SmokeHost SmokeObserver SmokeTests SmokeIdentifiers)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /WX /EHsc /permissive- /utf-8)
  target_link_libraries(${target} PRIVATE windowsapp ole32)
endforeach()
target_link_libraries(SmokeObserver PRIVATE gdi32)
target_link_libraries(SmokeHost PRIVATE shell32)
target_link_libraries(SmokeTests PRIVATE shell32)
foreach(target SmokeTask SmokeApp)
  target_compile_definitions(${target} PRIVATE SMOKE_APP WINAPI_FAMILY=WINAPI_FAMILY_APP)
  target_link_options(${target} PRIVATE /APPCONTAINER)
endforeach()
enable_testing()
add_test(NAME smoke-native COMMAND SmokeTests)
add_test(NAME smoke-identifiers COMMAND SmokeIdentifiers)
find_program(PWSH NAMES pwsh REQUIRED)
add_test(NAME smoke-wrapper COMMAND "${PWSH}" -NoProfile -File
  "${CMAKE_CURRENT_SOURCE_DIR}/tests/Wrapper.ps1" -Fixture $<TARGET_FILE:SmokeIdentifiers>)
set_tests_properties(smoke-native smoke-wrapper PROPERTIES TIMEOUT 20)
